```typescript
/**
 * Represents a symbolic value in execution
 */
export type SymbolicValue = 
    | ConcreteValue
    | SymbolicVariable
    | SymbolicExpression;

/**
 * A concrete (known) value
 */
export interface ConcreteValue {
    type: 'concrete';
    value: bigint;
    size: number;  // bits (8, 256, etc.)
}

/**
 * A symbolic variable (unknown value)
 */
export interface SymbolicVariable {
    type: 'symbolic';
    name: string;
    size: number;
    z3Var?: any;  // Z3 variable reference
}

/**
 * A symbolic expression (computation over symbolic values)
 */
export interface SymbolicExpression {
    type: 'expression';
    operator: Operator;
    operands: SymbolicValue[];
    size: number;
    z3Expr?: any;  // Cached Z3 expression
}

/**
 * All supported operators
 */
export type Operator =
    // Arithmetic
    | 'ADD' | 'SUB' | 'MUL' | 'DIV' | 'MOD'
    | 'ADDMOD' | 'MULMOD' | 'EXP'
    | 'SIGNEXTEND'
    
    // Comparison
    | 'LT' | 'GT' | 'SLT' | 'SGT'
    | 'EQ' | 'ISZERO'
    
    // Bitwise
    | 'AND' | 'OR' | 'XOR' | 'NOT'
    | 'BYTE' | 'SHL' | 'SHR' | 'SAR'
    
    // Memory/Storage access
    | 'MLOAD' | 'SLOAD'
    
    // Crypto
    | 'SHA3'
    
    // Conditional
    | 'ITE';  // If-then-else

/**
 * Helper functions for creating symbolic values
 */
export class SymbolicValueFactory {
    static concrete(value: bigint | number, size: number = 256): ConcreteValue {
        return {
            type: 'concrete',
            value: BigInt(value),
            size
        };
    }
    
    static symbolic(name: string, size: number = 256): SymbolicVariable {
        return {
            type: 'symbolic',
            name,
            size
        };
    }
    
    static expression(
        operator: Operator,
        operands: SymbolicValue[],
        size: number = 256
    ): SymbolicExpression {
        return {
            type: 'expression',
            operator,
            operands,
            size
        };
    }
    
    /**
     * Check if value is concrete
     */
    static isConcrete(value: SymbolicValue): value is ConcreteValue {
        return value.type === 'concrete';
    }
    
    /**
     * Check if value is fully concrete (no symbolic components)
     */
    static isFullyConcrete(value: SymbolicValue): boolean {
        if (value.type === 'concrete') return true;
        if (value.type === 'symbolic') return false;
        
        // Check expression operands recursively
        return value.operands.every(op => this.isFullyConcrete(op));
    }
    
    /**
     * Simplify symbolic value
     */
    static simplify(value: SymbolicValue): SymbolicValue {
        if (value.type !== 'expression') return value;
        
        // Simplify operands first
        const operands = value.operands.map(op => this.simplify(op));
        
        // If all operands are concrete, evaluate
        if (operands.every(op => op.type === 'concrete')) {
            return this.evaluate(value.operator, operands as ConcreteValue[]);
        }
        
        // Apply algebraic simplifications
        return this.algebraicSimplify(value.operator, operands);
    }
    
    /**
     * Evaluate concrete expression
     */
    private static evaluate(
        operator: Operator,
        operands: ConcreteValue[]
    ): ConcreteValue {
        const values = operands.map(op => op.value);
        
        switch (operator) {
            case 'ADD':
                return this.concrete(
                    (values[0] + values[1]) & ((1n << 256n) - 1n)
                );
            
            case 'SUB':
                return this.concrete(
                    (values[0] - values[1]) & ((1n << 256n) - 1n)
                );
            
            case 'MUL':
                return this.concrete(
                    (values[0] * values[1]) & ((1n << 256n) - 1n)
                );
            
            case 'DIV':
                return this.concrete(
                    values[1] === 0n ? 0n : values[0] / values[1]
                );
            
            case 'MOD':
                return this.concrete(
                    values[1] === 0n ? 0n : values[0] % values[1]
                );
            
            case 'LT':
                return this.concrete(values[0] < values[1] ? 1n : 0n);
            
            case 'GT':
                return this.concrete(values[0] > values[1] ? 1n : 0n);
            
            case 'EQ':
                return this.concrete(values[0] === values[1] ? 1n : 0n);
            
            case 'ISZERO':
                return this.concrete(values[0] === 0n ? 1n : 0n);
            
            case 'AND':
                return this.concrete(values[0] & values[1]);
            
            case 'OR':
                return this.concrete(values[0] | values[1]);
            
            case 'XOR':
                return this.concrete(values[0] ^ values[1]);
            
            case 'NOT':
                return this.concrete(~values[0] & ((1n << 256n) - 1n));
            
            // ... more operators
            
            default:
                throw new Error(`Unknown operator: ${operator}`);
        }
    }
    
    /**
     * Algebraic simplifications
     */
    private static algebraicSimplify(
        operator: Operator,
        operands: SymbolicValue[]
    ): SymbolicValue {
        // x + 0 = x
        if (operator === 'ADD' && 
            operands[1].type === 'concrete' && 
            operands[1].value === 0n) {
            return operands[0];
        }
        
        // x * 0 = 0
        if (operator === 'MUL' && 
            operands.some(op => op.type === 'concrete' && op.value === 0n)) {
            return this.concrete(0n);
        }
        
        // x * 1 = x
        if (operator === 'MUL' && 
            operands[1].type === 'concrete' && 
            operands[1].value === 1n) {
            return operands[0];
        }
        
        // x - x = 0
        if (operator === 'SUB' && this.areEqual(operands[0], operands[1])) {
            return this.concrete(0n);
        }
        
        // x & x = x
        if (operator === 'AND' && this.areEqual(operands[0], operands[1])) {
            return operands[0];
        }
        
        // x | 0 = x
        if (operator === 'OR' && 
            operands[1].type === 'concrete' && 
            operands[1].value === 0n) {
            return operands[0];
        }
        
        // ... many more simplifications
        
        // No simplification found
        return {
            type: 'expression',
            operator,
            operands,
            size: 256
        };
    }
    
    /**
     * Check if two symbolic values are structurally equal
     */
    private static areEqual(a: SymbolicValue, b: SymbolicValue): boolean {
        if (a.type !== b.type) return false;
        
        if (a.type === 'concrete' && b.type === 'concrete') {
            return a.value === b.value;
        }
        
        if (a.type === 'symbolic' && b.type === 'symbolic') {
            return a.name === b.name;
        }
        
        if (a.type === 'expression' && b.type === 'expression') {
            return a.operator === b.operator &&
                   a.operands.length === b.operands.length &&
                   a.operands.every((op, i) => this.areEqual(op, b.operands[i]));
        }
        
        return false;
    }
}
```

```typescript
/**
 * Represents a path constraint
 */
export interface Constraint {
    id: string;
    type: ConstraintType;
    expr: SymbolicValue;
    z3Expr?: any;  // Cached Z3 constraint
}

export type ConstraintType =
    | 'assert'      // Must be true
    | 'assume';     // Assumed to be true

export class ConstraintBuilder {
    private static nextId = 0;
    
    /**
     * Create assertion constraint
     */
    static assert(expr: SymbolicValue): Constraint {
        return {
            id: `c_${this.nextId++}`,
            type: 'assert',
            expr
        };
    }
    
    /**
     * Create assumption constraint
     */
    static assume(expr: SymbolicValue): Constraint {
        return {
            id: `c_${this.nextId++}`,
            type: 'assume',
            expr
        };
    }
    
    /**
     * Create constraint from comparison
     */
    static compare(
        lhs: SymbolicValue,
        op: 'EQ' | 'NEQ' | 'LT' | 'GT' | 'LTE' | 'GTE',
        rhs: SymbolicValue
    ): Constraint {
        let expr: SymbolicValue;
        
        switch (op) {
            case 'EQ':
                expr = SymbolicValueFactory.expression('EQ', [lhs, rhs]);
                break;
            case 'NEQ':
                expr = SymbolicValueFactory.expression(
                    'ISZERO',
                    [SymbolicValueFactory.expression('EQ', [lhs, rhs])]
                );
                break;
            case 'LT':
                expr = SymbolicValueFactory.expression('LT', [lhs, rhs]);
                break;
            case 'GT':
                expr = SymbolicValueFactory.expression('GT', [lhs, rhs]);
                break;
            case 'LTE':
                // lhs <= rhs  ≡  ¬(lhs > rhs)
                expr = SymbolicValueFactory.expression(
                    'ISZERO',
                    [SymbolicValueFactory.expression('GT', [lhs, rhs])]
                );
                break;
            case 'GTE':
                // lhs >= rhs  ≡  ¬(lhs < rhs)
                expr = SymbolicValueFactory.expression(
                    'ISZERO',
                    [SymbolicValueFactory.expression('LT', [lhs, rhs])]
                );
                break;
        }
        
        return this.assert(expr);
    }
}
```

```typescript
/**
 * Represents a complete execution path
 */
export interface ExecutionPath {
    id: string;
    constraints: Constraint[];
    state: SymbolicState;
    status: PathStatus;
    depth: number;          // Number of branches taken
    gasUsed: bigint;
    coverage: Set<number>;  // PCs visited
    parent?: string;        // Parent path ID
}

export type PathStatus =
    | 'active'      // Currently being explored
    | 'completed'   // Reached STOP/RETURN
    | 'reverted'    // Hit REVERT
    | 'invalid'     // Unsatisfiable constraints
    | 'timeout'     // Exceeded time/gas limit
    | 'error';      // Internal error

export class ExecutionPathFactory {
    private static nextId = 0;
    
    static create(state: SymbolicState): ExecutionPath {
        return {
            id: `path_${this.nextId++}`,
            constraints: [],
            state,
            status: 'active',
            depth: 0,
            gasUsed: 0n,
            coverage: new Set()
        };
    }
    
    /**
     * Fork path with additional constraint
     */
    static fork(
        parent: ExecutionPath,
        constraint: Constraint,
        state: SymbolicState
    ): ExecutionPath {
        return {
            id: `path_${this.nextId++}`,
            constraints: [...parent.constraints, constraint],
            state,
            status: 'active',
            depth: parent.depth + 1,
            gasUsed: parent.gasUsed,
            coverage: new Set(parent.coverage),
            parent: parent.id
        };
    }
}
```

```typescript
import { SymbolicValue, SymbolicValueFactory } from '../types/SymbolicValue';

/**
 * Represents the complete state of symbolic execution
 */
export class SymbolicState {
    // EVM state
    public stack: SymbolicValue[] = [];
    public memory: Map<bigint, SymbolicValue> = new Map();
    public storage: Map<bigint, SymbolicValue> = new Map();
    
    // Program counter
    public pc: number = 0;
    
    // Gas
    public gas: bigint;
    
    // Call context
    public calldata: SymbolicValue[];
    public caller: SymbolicValue;
    public callvalue: SymbolicValue;
    public address: SymbolicValue;
    public origin: SymbolicValue;
    
    // Block context
    public timestamp: SymbolicValue;
    public number: SymbolicValue;
    public difficulty: SymbolicValue;
    public gaslimit: SymbolicValue;
    public coinbase: SymbolicValue;
    public chainid: SymbolicValue;
    
    // Return data
    public returnData: SymbolicValue[] = [];
    
    constructor(
        gas: bigint = 1_000_000n,
        calldata: SymbolicValue[] = [],
        makeSymbolic: boolean = true
    ) {
        this.gas = gas;
        this.calldata = calldata;
        
        if (makeSymbolic) {
            // Create symbolic environment
            this.caller = SymbolicValueFactory.symbolic('caller', 160);
            this.callvalue = SymbolicValueFactory.symbolic('callvalue');
            this.address = SymbolicValueFactory.symbolic('address', 160);
            this.origin = SymbolicValueFactory.symbolic('origin', 160);
            this.timestamp = SymbolicValueFactory.symbolic('timestamp');
            this.number = SymbolicValueFactory.symbolic('number');
            this.difficulty = SymbolicValueFactory.symbolic('difficulty');
            this.gaslimit = SymbolicValueFactory.symbolic('gaslimit');
            this.coinbase = SymbolicValueFactory.symbolic('coinbase', 160);
            this.chainid = SymbolicValueFactory.concrete(1n);  // Mainnet
        } else {
            // Concrete values for testing
            this.caller = SymbolicValueFactory.concrete(0n, 160);
            this.callvalue = SymbolicValueFactory.concrete(0n);
            this.address = SymbolicValueFactory.concrete(0n, 160);
            this.origin = SymbolicValueFactory.concrete(0n, 160);
            this.timestamp = SymbolicValueFactory.concrete(0n);
            this.number = SymbolicValueFactory.concrete(0n);
            this.difficulty = SymbolicValueFactory.concrete(0n);
            this.gaslimit = SymbolicValueFactory.concrete(30_000_000n);
            this.coinbase = SymbolicValueFactory.concrete(0n, 160);
            this.chainid = SymbolicValueFactory.concrete(1n);
        }
    }
    
    /**
     * Deep clone the state
     */
    clone(): SymbolicState {
        const cloned = new SymbolicState(this.gas, [], false);
        
        // Clone stack
        cloned.stack = [...this.stack];
        
        // Clone memory
        cloned.memory = new Map(this.memory);
        
        // Clone storage
        cloned.storage = new Map(this.storage);
        
        // Clone context
        cloned.pc = this.pc;
        cloned.calldata = [...this.calldata];
        cloned.caller = this.caller;
        cloned.callvalue = this.callvalue;
        cloned.address = this.address;
        cloned.origin = this.origin;
        cloned.timestamp = this.timestamp;
        cloned.number = this.number;
        cloned.difficulty = this.difficulty;
        cloned.gaslimit = this.gaslimit;
        cloned.coinbase = this.coinbase;
        cloned.chainid = this.chainid;
        cloned.returnData = [...this.returnData];
        
        return cloned;
    }
    
    /**
     * Stack operations
     */
    push(value: SymbolicValue): void {
        if (this.stack.length >= 1024) {
            throw new Error('Stack overflow');
        }
        this.stack.push(value);
    }
    
    pop(): SymbolicValue {
        if (this.stack.length === 0) {
            throw new Error('Stack underflow');
        }
        return this.stack.pop()!;
    }
    
    peek(index: number = 0): SymbolicValue {
        if (index >= this.stack.length) {
            throw new Error('Stack underflow');
        }
        return this.stack[this.stack.length - 1 - index];
    }
    
    swap(n: number): void {
        if (n >= this.stack.length) {
            throw new Error('Stack underflow');
        }
        const top = this.stack.length - 1;
        const temp = this.stack[top];
        this.stack[top] = this.stack[top - n];
        this.stack[top - n] = temp;
    }
    
    dup(n: number): void {
        if (n > this.stack.length) {
            throw new Error('Stack underflow');
        }
        const value = this.stack[this.stack.length - n];
        this.push(value);
    }
    
    /**
     * Memory operations
     */
    mload(offset: SymbolicValue): SymbolicValue {
        // Simplified: assume 32-byte aligned reads
        if (offset.type === 'concrete') {
            const addr = offset.value;
            
            // Return stored value or zero
            return this.memory.get(addr) || SymbolicValueFactory.concrete(0n);
        }
        
        // Symbolic offset - return fresh symbolic variable
        return SymbolicValueFactory.symbolic(`mem[${this.stringifyValue(offset)}]`);
    }
    
    mstore(offset: SymbolicValue, value: SymbolicValue): void {
        if (offset.type === 'concrete') {
            this.memory.set(offset.value, value);
        } else {
            // Symbolic offset - havoc memory (conservative)
            console.warn('Symbolic memory write - havocing memory');
        }
    }
    
    /**
     * Storage operations
     */
    sload(key: SymbolicValue): SymbolicValue {
        if (key.type === 'concrete') {
            return this.storage.get(key.value) || SymbolicValueFactory.concrete(0n);
        }
        
        // Symbolic key - return fresh symbolic variable
        return SymbolicValueFactory.symbolic(`storage[${this.stringifyValue(key)}]`);
    }
    
    sstore(key: SymbolicValue, value: SymbolicValue): void {
        if (key.type === 'concrete') {
            this.storage.set(key.value, value);
        } else {
            console.warn('Symbolic storage write');
        }
    }
    
    /**
     * Helper to stringify symbolic value for debugging
     */
    private stringifyValue(value: SymbolicValue): string {
        if (value.type === 'concrete') {
            return `0x${value.value.toString(16)}`;
        }
        if (value.type === 'symbolic') {
            return value.name;
        }
        return `${value.operator}(...)`;
    }
}
```

```typescript
/**
 * Configuration for symbolic execution
 */
export interface SymbolicVMConfig {
    maxPaths: number;
    maxDepth: number;
    timeout: number;          // milliseconds
    checkOverflow: boolean;
    checkReentrancy: boolean;
    checkAccessControl: boolean;
    explorationStrategy: 'dfs' | 'bfs' | 'random' | 'coverage';
}

const DEFAULT_CONFIG: SymbolicVMConfig = {
    maxPaths: 1000,
    maxDepth: 100,
    timeout: 60_000,
    checkOverflow: true,
    checkReentrancy: true,
    checkAccessControl: true,
    explorationStrategy: 'coverage'
};

/**
 * Main symbolic execution engine
 */
export class SymbolicVM {
    private bytecode: Uint8Array;
    private config: SymbolicVMConfig;
    private solver: ConstraintSolver;
    private detector: VulnerabilityDetector;
    
    private worklist: ExecutionPath[] = [];
    private completedPaths: ExecutionPath[] = [];
    private coverage: Set<number> = new Set();
    private startTime: number = 0;
    
    // Valid jump destinations
    private jumpDests: Set<number> = new Set();
    
    constructor(
        bytecode: string,
        config: Partial<SymbolicVMConfig> = {}
    ) {
        // Parse bytecode
        this.bytecode = this.parseBytecode(bytecode);
        this.config = { ...DEFAULT_CONFIG, ...config };
        this.solver = new ConstraintSolver();
        this.detector = new VulnerabilityDetector(this.solver);
        
        // Find all JUMPDEST locations
        this.findJumpDests();
    }
    
    private parseBytecode(bytecode: string): Uint8Array {
        const hex = bytecode.startsWith('0x') ? bytecode.slice(2) : bytecode;
        const bytes = new Uint8Array(hex.length / 2);
        for (let i = 0; i < bytes.length; i++) {
            bytes[i] = parseInt(hex.slice(i * 2, i * 2 + 2), 16);
        }
        return bytes;
    }
    
    private findJumpDests(): void {
        let pc = 0;
        while (pc < this.bytecode.length) {
            const opcode = this.bytecode[pc];
            
            if (opcode === 0x5B) {  // JUMPDEST
                this.jumpDests.add(pc);
            }
            
            // Skip PUSH operands
            if (opcode >= 0x60 && opcode <= 0x7F) {
                pc += opcode - 0x60 + 2;  // opcode + n bytes
            } else {
                pc += 1;
            }
        }
    }
    
    /**
     * Run symbolic execution
     */
    async execute(): Promise<ExecutionResult> {
        this.startTime = Date.now();
        
        // Create initial state
        const initialState = new SymbolicState();
        const initialPath = ExecutionPathFactory.create(initialState);
        this.worklist.push(initialPath);
        
        // Main execution loop
        while (this.worklist.length > 0) {
            // Check timeout
            if (Date.now() - this.startTime > this.config.timeout) {
                break;
            }
            
            // Check path limit
            if (this.completedPaths.length >= this.config.maxPaths) {
                break;
            }
            
            // Get next path based on exploration strategy
            const path = this.selectNextPath();
            if (!path) break;
            
            // Execute one step
            await this.step(path);
        }
        
        // Run vulnerability detection
        const vulnerabilities = await this.detector.analyze(this.completedPaths);
        
        return {
            completedPaths: this.completedPaths.length,
            coverage: this.coverage.size,
            coveragePercent: this.calculateCoveragePercent(),
            vulnerabilities,
            timeElapsed: Date.now() - this.startTime
        };
    }
    
    private selectNextPath(): ExecutionPath | null {
        switch (this.config.explorationStrategy) {
            case 'dfs':
                return this.worklist.pop() || null;
            
            case 'bfs':
                return this.worklist.shift() || null;
            
            case 'random':
                if (this.worklist.length === 0) return null;
                const idx = Math.floor(Math.random() * this.worklist.length);
                return this.worklist.splice(idx, 1)[0];
            
            case 'coverage':
                return this.selectCoverageGuided();
            
            default:
                return this.worklist.pop() || null;
        }
    }
    
    private selectCoverageGuided(): ExecutionPath | null {
        if (this.worklist.length === 0) return null;
        
        // Find path with most new coverage potential
        let bestPath: ExecutionPath | null = null;
        let bestScore = -1;
        let bestIdx = -1;
        
        for (let i = 0; i < this.worklist.length; i++) {
            const path = this.worklist[i];
            const pc = path.state.pc;
            
            // Score based on uncovered code at current PC
            let score = this.coverage.has(pc) ? 0 : 1;
            
            // Bonus for paths closer to uncovered code
            score += this.estimateNewCoverage(path);
            
            if (score > bestScore) {
                bestScore = score;
                bestPath = path;
                bestIdx = i;
            }
        }
        
        if (bestIdx >= 0) {
            this.worklist.splice(bestIdx, 1);
        }
        
        return bestPath;
    }
    
    private estimateNewCoverage(path: ExecutionPath): number {
        // Simple heuristic: count PCs near current that aren't covered
        let score = 0;
        for (let offset = 1; offset <= 10; offset++) {
            if (!this.coverage.has(path.state.pc + offset)) {
                score += 1 / offset;
            }
        }
        return score;
    }
    
    /**
     * Execute one instruction
     */
    private async step(path: ExecutionPath): Promise<void> {
        const state = path.state;
        const pc = state.pc;
        
        // Check bounds
        if (pc >= this.bytecode.length) {
            path.status = 'completed';
            this.completedPaths.push(path);
            return;
        }
        
        // Track coverage
        this.coverage.add(pc);
        path.coverage.add(pc);
        
        // Get opcode
        const opcode = this.bytecode[pc];
        const opcodeInfo = OPCODES[opcode];
        
        if (!opcodeInfo) {
            path.status = 'error';
            this.completedPaths.push(path);
            return;
        }
        
        // Execute based on opcode
        try {
            switch (opcodeInfo.name) {
                // Arithmetic
                case 'ADD':
                    this.executeADD(state);
                    break;
                case 'SUB':
                    this.executeSUB(state, path);
                    break;
                case 'MUL':
                    this.executeMUL(state);
                    break;
                case 'DIV':
                    this.executeDIV(state);
                    break;
                case 'MOD':
                    this.executeMOD(state);
                    break;
                
                // Comparison
                case 'LT':
                    this.executeLT(state);
                    break;
                case 'GT':
                    this.executeGT(state);
                    break;
                case 'EQ':
                    this.executeEQ(state);
                    break;
                case 'ISZERO':
                    this.executeISZERO(state);
                    break;
                
                // Bitwise
                case 'AND':
                    this.executeAND(state);
                    break;
                case 'OR':
                    this.executeOR(state);
                    break;
                case 'XOR':
                    this.executeXOR(state);
                    break;
                case 'NOT':
                    this.executeNOT(state);
                    break;
                case 'SHL':
                    this.executeSHL(state);
                    break;
                case 'SHR':
                    this.executeSHR(state);
                    break;
                
                // Memory
                case 'MLOAD':
                    this.executeMLOAD(state);
                    break;
                case 'MSTORE':
                    this.executeMSTORE(state);
                    break;
                case 'MSTORE8':
                    this.executeMSTORE8(state);
                    break;
                
                // Storage
                case 'SLOAD':
                    this.executeSLOAD(state);
                    break;
                case 'SSTORE':
                    this.executeSSTORE(state, path);
                    break;
                
                // Control flow
                case 'JUMP':
                    await this.executeJUMP(state, path);
                    return;  // Don't increment PC
                case 'JUMPI':
                    await this.executeJUMPI(state, path);
                    return;  // Don't increment PC
                case 'JUMPDEST':
                    // No-op, just a marker
                    break;
                
                // Environment
                case 'ADDRESS':
                    state.push(state.address);
                    break;
                case 'CALLER':
                    state.push(state.caller);
                    break;
                case 'CALLVALUE':
                    state.push(state.callvalue);
                    break;
                case 'CALLDATALOAD':
                    this.executeCALLDATALOAD(state);
                    break;
                case 'CALLDATASIZE':
                    state.push(SymbolicValueFactory.symbolic('calldatasize'));
                    break;
                case 'TIMESTAMP':
                    state.push(state.timestamp);
                    break;
                case 'NUMBER':
                    state.push(state.number);
                    break;
                
                // Stack
                case 'POP':
                    state.pop();
                    break;
                case 'PC':
                    state.push(SymbolicValueFactory.concrete(BigInt(pc)));
                    break;
                case 'GAS':
                    state.push(SymbolicValueFactory.symbolic('gas'));
                    break;
                
                // PUSH
                default:
                    if (opcodeInfo.name.startsWith('PUSH')) {
                        const n = parseInt(opcodeInfo.name.slice(4));
                        const value = this.readPushValue(pc + 1, n);
                        state.push(SymbolicValueFactory.concrete(value));
                        state.pc = pc + 1 + n;
                        this.worklist.push(path);
                        return;
                    }
                    
                    if (opcodeInfo.name.startsWith('DUP')) {
                        const n = parseInt(opcodeInfo.name.slice(3));
                        state.dup(n);
                        break;
                    }
                    
                    if (opcodeInfo.name.startsWith('SWAP')) {
                        const n = parseInt(opcodeInfo.name.slice(4));
                        state.swap(n);
                        break;
                    }
                    
                    // Terminal opcodes
                    if (opcodeInfo.name === 'STOP' || 
                        opcodeInfo.name === 'RETURN') {
                        path.status = 'completed';
                        this.completedPaths.push(path);
                        return;
                    }
                    
                    if (opcodeInfo.name === 'REVERT') {
                        path.status = 'reverted';
                        this.completedPaths.push(path);
                        return;
                    }
                    
                    if (opcodeInfo.name === 'INVALID') {
                        path.status = 'error';
                        this.completedPaths.push(path);
                        return;
                    }
                    
                    if (opcodeInfo.name === 'SELFDESTRUCT') {
                        this.detector.checkSelfDestruct(path);
                        path.status = 'completed';
                        this.completedPaths.push(path);
                        return;
                    }
                    
                    // CALL variants
                    if (opcodeInfo.name === 'CALL' ||
                        opcodeInfo.name === 'DELEGATECALL' ||
                        opcodeInfo.name === 'STATICCALL') {
                        await this.executeCALL(state, path, opcodeInfo.name);
                        break;
                    }
                    
                    // Unhandled opcode
                    console.warn(`Unhandled opcode: ${opcodeInfo.name}`);
            }
            
            // Increment PC and continue
            state.pc = pc + 1;
            this.worklist.push(path);
            
        } catch (error) {
            path.status = 'error';
            this.completedPaths.push(path);
        }
    }
    
    // Arithmetic implementations
    private executeADD(state: SymbolicState): void {
        const b = state.pop();
        const a = state.pop();
        
        if (a.type === 'concrete' && b.type === 'concrete') {
            const result = (a.value + b.value) & ((1n << 256n) - 1n);
            state.push(SymbolicValueFactory.concrete(result));
        } else {
            state.push(SymbolicValueFactory.expression('ADD', [a, b]));
        }
    }
    
    private executeSUB(state: SymbolicState, path: ExecutionPath): void {
        const b = state.pop();
        const a = state.pop();
        
        // Check for underflow
        if (this.config.checkOverflow) {
            const underflowConstraint = ConstraintBuilder.compare(b, 'GT', a);
            // Note: We don't block execution, just record potential issue
            if (a.type !== 'concrete' || b.type !== 'concrete' || b.value > a.value) {
                this.detector.recordPotentialUnderflow(path, state.pc, a, b);
            }
        }
        
        if (a.type === 'concrete' && b.type === 'concrete') {
            const result = (a.value - b.value) & ((1n << 256n) - 1n);
            state.push(SymbolicValueFactory.concrete(result));
        } else {
            state.push(SymbolicValueFactory.expression('SUB', [a, b]));
        }
    }
    
    private executeMUL(state: SymbolicState): void {
        const b = state.pop();
        const a = state.pop();
        
        if (a.type === 'concrete' && b.type === 'concrete') {
            const result = (a.value * b.value) & ((1n << 256n) - 1n);
            state.push(SymbolicValueFactory.concrete(result));
        } else {
            state.push(SymbolicValueFactory.expression('MUL', [a, b]));
        }
    }
    
    private executeDIV(state: SymbolicState): void {
        const b = state.pop();
        const a = state.pop();
        
        if (a.type === 'concrete' && b.type === 'concrete') {
            const result = b.value === 0n ? 0n : a.value / b.value;
            state.push(SymbolicValueFactory.concrete(result));
        } else {
            state.push(SymbolicValueFactory.expression('DIV', [a, b]));
        }
    }
    
    private executeMOD(state: SymbolicState): void {
        const b = state.pop();
        const a = state.pop();
        
        if (a.type === 'concrete' && b.type === 'concrete') {
            const result = b.value === 0n ? 0n : a.value % b.value;
            state.push(SymbolicValueFactory.concrete(result));
        } else {
            state.push(SymbolicValueFactory.expression('MOD', [a, b]));
        }
    }
    
    // Comparison implementations
    private executeLT(state: SymbolicState): void {
        const b = state.pop();
        const a = state.pop();
        
        if (a.type === 'concrete' && b.type === 'concrete') {
            state.push(SymbolicValueFactory.concrete(a.value < b.value ? 1n : 0n));
        } else {
            state.push(SymbolicValueFactory.expression('LT', [a, b]));
        }
    }
    
    private executeGT(state: SymbolicState): void {
        const b = state.pop();
        const a = state.pop();
        
        if (a.type === 'concrete' && b.type === 'concrete') {
            state.push(SymbolicValueFactory.concrete(a.value > b.value ? 1n : 0n));
        } else {
            state.push(SymbolicValueFactory.expression('GT', [a, b]));
        }
    }
    
    private executeEQ(state: SymbolicState): void {
        const b = state.pop();
        const a = state.pop();
        
        if (a.type === 'concrete' && b.type === 'concrete') {
            state.push(SymbolicValueFactory.concrete(a.value === b.value ? 1n : 0n));
        } else {
            state.push(SymbolicValueFactory.expression('EQ', [a, b]));
        }
    }
    
    private executeISZERO(state: SymbolicState): void {
        const a = state.pop();
        
        if (a.type === 'concrete') {
            state.push(SymbolicValueFactory.concrete(a.value === 0n ? 1n : 0n));
        } else {
            state.push(SymbolicValueFactory.expression('ISZERO', [a]));
        }
    }
    
    // Bitwise implementations
    private executeAND(state: SymbolicState): void {
        const b = state.pop();
        const a = state.pop();
        
        if (a.type === 'concrete' && b.type === 'concrete') {
            state.push(SymbolicValueFactory.concrete(a.value & b.value));
        } else {
            state.push(SymbolicValueFactory.expression('AND', [a, b]));
        }
    }
    
    private executeOR(state: SymbolicState): void {
        const b = state.pop();
        const a = state.pop();
        
        if (a.type === 'concrete' && b.type === 'concrete') {
            state.push(SymbolicValueFactory.concrete(a.value | b.value));
        } else {
            state.push(SymbolicValueFactory.expression('OR', [a, b]));
        }
    }
    
    private executeXOR(state: SymbolicState): void {
        const b = state.pop();
        const a = state.pop();
        
        if (a.type === 'concrete' && b.type === 'concrete') {
            state.push(SymbolicValueFactory.concrete(a.value ^ b.value));
        } else {
            state.push(SymbolicValueFactory.expression('XOR', [a, b]));
        }
    }
    
    private executeNOT(state: SymbolicState): void {
        const a = state.pop();
        
        if (a.type === 'concrete') {
            const result = ~a.value & ((1n << 256n) - 1n);
            state.push(SymbolicValueFactory.concrete(result));
        } else {
            state.push(SymbolicValueFactory.expression('NOT', [a]));
        }
    }
    
    private executeSHL(state: SymbolicState): void {
        const shift = state.pop();
        const value = state.pop();
        
        if (shift.type === 'concrete' && value.type === 'concrete') {
            const result = shift.value >= 256n 
                ? 0n 
                : (value.value << shift.value) & ((1n << 256n) - 1n);
            state.push(SymbolicValueFactory.concrete(result));
        } else {
            state.push(SymbolicValueFactory.expression('SHL', [shift, value]));
        }
    }
    
    private executeSHR(state: SymbolicState): void {
        const shift = state.pop();
        const value = state.pop();
        
        if (shift.type === 'concrete' && value.type === 'concrete') {
            const result = shift.value >= 256n ? 0n : value.value >> shift.value;
            state.push(SymbolicValueFactory.concrete(result));
        } else {
            state.push(SymbolicValueFactory.expression('SHR', [shift, value]));
        }
    }
    
    // Memory implementations
    private executeMLOAD(state: SymbolicState): void {
        const offset = state.pop();
        state.push(state.mload(offset));
    }
    
    private executeMSTORE(state: SymbolicState): void {
        const offset = state.pop();
        const value = state.pop();
        state.mstore(offset, value);
    }
    
    private executeMSTORE8(state: SymbolicState): void {
        const offset = state.pop();
        const value = state.pop();
        // Store only the lowest byte
        if (value.type === 'concrete') {
            const byte = SymbolicValueFactory.concrete(value.value & 0xFFn);
            state.mstore(offset, byte);
        } else {
            const byte = SymbolicValueFactory.expression('AND', [
                value,
                SymbolicValueFactory.concrete(0xFFn)
            ]);
            state.mstore(offset, byte);
        }
    }
    
    // Storage implementations
    private executeSLOAD(state: SymbolicState): void {
        const key = state.pop();
        state.push(state.sload(key));
    }
    
    private executeSSTORE(state: SymbolicState, path: ExecutionPath): void {
        const key = state.pop();
        const value = state.pop();
        
        // Record for reentrancy detection
        this.detector.recordStorageWrite(path, state.pc, key, value);
        
        state.sstore(key, value);
    }
    
    // Control flow implementations
    private async executeJUMP(state: SymbolicState, path: ExecutionPath): Promise<void> {
        const dest = state.pop();
        
        if (dest.type === 'concrete') {
            const target = Number(dest.value);
            
            // Validate jump destination
            if (!this.jumpDests.has(target)) {
                path.status = 'error';
                this.completedPaths.push(path);
                return;
            }
            
            state.pc = target;
            this.worklist.push(path);
        } else {
            // Symbolic jump - try all valid destinations
            for (const target of this.jumpDests) {
                const branchPath = ExecutionPathFactory.fork(
                    path,
                    ConstraintBuilder.compare(
                        dest,
                        'EQ',
                        SymbolicValueFactory.concrete(BigInt(target))
                    ),
                    state.clone()
                );
                
                // Check if path is satisfiable
                if (await this.solver.isSatisfiable(branchPath.constraints)) {
                    branchPath.state.pc = target;
                    this.worklist.push(branchPath);
                }
            }
        }
    }
    
    private async executeJUMPI(state: SymbolicState, path: ExecutionPath): Promise<void> {
        const dest = state.pop();
        const condition = state.pop();
        
        // Check if condition is concrete
        if (condition.type === 'concrete') {
            if (condition.value !== 0n) {
                // Take jump
                if (dest.type === 'concrete') {
                    const target = Number(dest.value);
                    if (!this.jumpDests.has(target)) {
                        path.status = 'error';
                        this.completedPaths.push(path);
                        return;
                    }
                    state.pc = target;
                } else {
                    // Symbolic destination with concrete true condition
                    path.status = 'error';
                    this.completedPaths.push(path);
                    return;
                }
            } else {
                // Fall through
                state.pc = state.pc + 1;
            }
            this.worklist.push(path);
            return;
        }
        
        // Symbolic condition - fork execution
        const pc = state.pc;
        
        // True branch (take jump)
        if (dest.type === 'concrete') {
            const target = Number(dest.value);
            if (this.jumpDests.has(target)) {
                const trueConstraint = ConstraintBuilder.compare(
                    condition,
                    'NEQ',
                    SymbolicValueFactory.concrete(0n)
                );
                
                const trueState = state.clone();
                trueState.pc = target;
                
                const truePath = ExecutionPathFactory.fork(path, trueConstraint, trueState);
                
                if (await this.solver.isSatisfiable(truePath.constraints)) {
                    this.worklist.push(truePath);
                }
            }
        }
        
        // False branch (fall through)
        const falseConstraint = ConstraintBuilder.compare(
            condition,
            'EQ',
            SymbolicValueFactory.concrete(0n)
        );
        
        const falseState = state.clone();
        falseState.pc = pc + 1;
        
        const falsePath = ExecutionPathFactory.fork(path, falseConstraint, falseState);
        
        if (await this.solver.isSatisfiable(falsePath.constraints)) {
            this.worklist.push(falsePath);
        }
    }
    
    // Environment implementations
    private executeCALLDATALOAD(state: SymbolicState): void {
        const offset = state.pop();
        
        if (offset.type === 'concrete') {
            const idx = Number(offset.value);
            if (idx < state.calldata.length) {
                state.push(state.calldata[idx]);
            } else {
                // Return symbolic value for unknown calldata
                state.push(SymbolicValueFactory.symbolic(`calldata[${idx}]`));
            }
        } else {
            // Symbolic offset
            state.push(SymbolicValueFactory.symbolic(`calldata[${state.stringifyValue(offset)}]`));
        }
    }
    
    // CALL implementation
    private async executeCALL(
        state: SymbolicState,
        path: ExecutionPath,
        callType: string
    ): Promise<void> {
        // Pop arguments
        const gas = state.pop();
        const addr = state.pop();
        const value = callType === 'CALL' ? state.pop() : SymbolicValueFactory.concrete(0n);
        const argsOffset = state.pop();
        const argsSize = state.pop();
        const retOffset = state.pop();
        const retSize = state.pop();
        
        // Record for reentrancy detection
        this.detector.recordExternalCall(path, state.pc, addr, value, callType);
        
        // Push success value (symbolic - we don't know the result)
        state.push(SymbolicValueFactory.symbolic(`${callType}_success_${state.pc}`));
        
        // Set return data to symbolic
        state.returnData = [SymbolicValueFactory.symbolic(`${callType}_returndata_${state.pc}`)];
    }
    
    // Helper to read PUSH operand
    private readPushValue(start: number, n: number): bigint {
        let value = 0n;
        for (let i = 0; i < n; i++) {
            if (start + i < this.bytecode.length) {
                value = (value << 8n) | BigInt(this.bytecode[start + i]);
            }
        }
        return value;
    }
    
    private calculateCoveragePercent(): number {
        // Calculate code coverage as percentage
        let codeBytes = 0;
        let pc = 0;
        
        while (pc < this.bytecode.length) {
            codeBytes++;
            const opcode = this.bytecode[pc];
            if (opcode >= 0x60 && opcode <= 0x7F) {
                pc += opcode - 0x60 + 2;
            } else {
                pc++;
            }
        }
        
        return (this.coverage.size / codeBytes) * 100;
    }
}

export interface ExecutionResult {
```

```typescript
import { init as initZ3 } from 'z3-solver';
import { SymbolicValue } from '../types/SymbolicValue';
import { Constraint } from '../types/Constraint';

/**
 * Z3 constraint solver wrapper
 */
export class ConstraintSolver {
    private z3: any;
    private ctx: any;
    private solver: any;
    private initialized: boolean = false;
    
    async initialize(): Promise<void> {
        if (this.initialized) return;
        
        const { Context } = await initZ3();
        this.ctx = new Context('main');
        this.solver = new this.ctx.Solver();
        this.initialized = true;
    }
    
    /**
     * Check if constraints are satisfiable
     */
    async isSatisfiable(constraints: Constraint[]): Promise<boolean> {
        await this.initialize();
        
        this.solver.push();
        
        try {
            for (const constraint of constraints) {
                const z3Expr = this.toZ3(constraint.expr);
                this.solver.add(z3Expr);
            }
            
            const result = await this.solver.check();
            return result === 'sat';
        } finally {
            this.solver.pop();
        }
    }
    
    /**
     * Get a model (concrete solution) for constraints
     */
    async getModel(constraints: Constraint[]): Promise<Map<string, bigint> | null> {
        await this.initialize();
        
        this.solver.push();
        
        try {
            for (const constraint of constraints) {
                const z3Expr = this.toZ3(constraint.expr);
                this.solver.add(z3Expr);
            }
            
            const result = await this.solver.check();
            
            if (result !== 'sat') {
                return null;
            }
            
            const model = this.solver.model();
            const values = new Map<string, bigint>();
            
            for (const decl of model.decls()) {
                const name = decl.name();
                const value = model.eval(decl());
                values.set(name, BigInt(value.toString()));
            }
            
            return values;
        } finally {
            this.solver.pop();
        }
    }
    
    /**
     * Convert symbolic value to Z3 expression
     */
    private toZ3(value: SymbolicValue): any {
        const { BitVec } = this.ctx;
        
        switch (value.type) {
            case 'concrete':
                return BitVec.val(value.value, 256);
            
            case 'symbolic':
                // Get or create Z3 variable
                if (!value.z3Var) {
                    value.z3Var = BitVec.const(value.name, value.size);
                }
                return value.z3Var;
            
            case 'expression':
                return this.expressionToZ3(value);
        }
    }
    
    /**
     * Convert symbolic expression to Z3
     */
    private expressionToZ3(expr: any): any {
        const { BitVec } = this.ctx;
        
        const operands = expr.operands.map((op: SymbolicValue) => this.toZ3(op));
        
        switch (expr.operator) {
            case 'ADD':
                return operands[0].add(operands[1]);
            case 'SUB':
                return operands[0].sub(operands[1]);
            case 'MUL':
                return operands[0].mul(operands[1]);
            case 'DIV':
                const zero = BitVec.val(0, 256);
                return operands[1].eq(zero).ite(zero, operands[0].udiv(operands[1]));
            case 'MOD':
                const zero2 = BitVec.val(0, 256);
                return operands[1].eq(zero2).ite(zero2, operands[0].umod(operands[1]));
            
            case 'LT':
                return operands[0].ult(operands[1]).ite(
                    BitVec.val(1, 256),
                    BitVec.val(0, 256)
                );
            case 'GT':
                return operands[0].ugt(operands[1]).ite(
                    BitVec.val(1, 256),
                    BitVec.val(0, 256)
                );
            case 'EQ':
                return operands[0].eq(operands[1]).ite(
                    BitVec.val(1, 256),
                    BitVec.val(0, 256)
                );
            case 'ISZERO':
                return operands[0].eq(BitVec.val(0, 256)).ite(
                    BitVec.val(1, 256),
                    BitVec.val(0, 256)
                );
            
            case 'AND':
                return operands[0].and(operands[1]);
            case 'OR':
                return operands[0].or(operands[1]);
            case 'XOR':
                return operands[0].xor(operands[1]);
            case 'NOT':
                return operands[0].not();
            case 'SHL':
                return operands[1].shl(operands[0]);
            case 'SHR':
                return operands[1].lshr(operands[0]);
            
            default:
                throw new Error(`Unknown operator: ${expr.operator}`);
        }
    }
}
```

## 8.1 The Vulnerability Detector

**src/analysis/VulnerabilityDetector.ts:**

```typescript
import { ExecutionPath } from '../types/ExecutionPath';
import { SymbolicValue } from '../types/SymbolicValue';
import { ConstraintSolver } from '../core/ConstraintSolver';
import { Vulnerability, ConcreteInput } from '../core/SymbolicVM';

interface ExternalCallRecord {
    pc: number;
    address: SymbolicValue;
    value: SymbolicValue;
    callType: string;
    pathId: string;
}

interface StorageWriteRecord {
    pc: number;
    key: SymbolicValue;
    value: SymbolicValue;
    pathId: string;
}

interface OverflowRecord {
    pc: number;
    operand1: SymbolicValue;
    operand2: SymbolicValue;
    operation: 'ADD' | 'SUB' | 'MUL';
    pathId: string;
}

export class VulnerabilityDetector {
    private solver: ConstraintSolver;
    private externalCalls: Map<string, ExternalCallRecord[]> = new Map();
    private storageWrites: Map<string, StorageWriteRecord[]> = new Map();
    private potentialOverflows: OverflowRecord[] = [];
    
    constructor(solver: ConstraintSolver) {
        this.solver = solver;
    }
    
    recordExternalCall(
        path: ExecutionPath,
        pc: number,
        address: SymbolicValue,
        value: SymbolicValue,
        callType: string
    ): void {
        const records = this.externalCalls.get(path.id) || [];
        records.push({ pc, address, value, callType, pathId: path.id });
        this.externalCalls.set(path.id, records);
    }
    
    recordStorageWrite(
        path: ExecutionPath,
        pc: number,
        key: SymbolicValue,
        value: SymbolicValue
    ): void {
        const records = this.storageWrites.get(path.id) || [];
        records.push({ pc, key, value, pathId: path.id });
        this.storageWrites.set(path.id, records);
    }
    
    recordPotentialUnderflow(
        path: ExecutionPath,
        pc: number,
        operand1: SymbolicValue,
        operand2: SymbolicValue
    ): void {
        this.potentialOverflows.push({
            pc,
            operand1,
            operand2,
            operation: 'SUB',
            pathId: path.id
        });
    }
    
    checkSelfDestruct(path: ExecutionPath): void {
        // Self-destruct found - will be reported in analyze
    }
    
    async analyze(paths: ExecutionPath[]): Promise<Vulnerability[]> {
        const vulnerabilities: Vulnerability[] = [];
        
        // Detect reentrancy
        for (const path of paths) {
            const reentVulns = await this.detectReentrancy(path);
            vulnerabilities.push(...reentVulns);
        }
        
        // Detect overflows/underflows
        const overflowVulns = await this.detectOverflows(paths);
        vulnerabilities.push(...overflowVulns);
        
        // Detect self-destruct issues
        for (const path of paths) {
            const sdVulns = await this.detectSelfDestruct(path);
            vulnerabilities.push(...sdVulns);
        }
        
        return vulnerabilities;
    }
    
    private async detectReentrancy(path: ExecutionPath): Promise<Vulnerability[]> {
        const vulnerabilities: Vulnerability[] = [];
        
        const calls = this.externalCalls.get(path.id) || [];
        const writes = this.storageWrites.get(path.id) || [];
        
        // Check each external call
        for (const call of calls) {
            // Find storage writes AFTER this call
            const writesAfter = writes.filter(w => w.pc > call.pc);
            
            if (writesAfter.length > 0) {
                // Potential reentrancy: external call before state update
                
                // Try to generate exploit
                const exploit = await this.generateExploit(path);
                
                vulnerabilities.push({
                    type: 'REENTRANCY',
                    severity: 'HIGH',
                    pc: call.pc,
                    description: `External ${call.callType} at PC ${call.pc} before storage write at PC ${writesAfter[0].pc}. Classic reentrancy pattern detected.`,
                    path: path.constraints,
                    exploit
                });
            }
        }
        
        return vulnerabilities;
    }
    
    private async detectOverflows(paths: ExecutionPath[]): Promise<Vulnerability[]> {
        const vulnerabilities: Vulnerability[] = [];
        
        for (const record of this.potentialOverflows) {
            const path = paths.find(p => p.id === record.pathId);
            if (!path) continue;
            
            // Check if overflow is actually reachable
            const canOverflow = await this.solver.isSatisfiable([
                ...path.constraints,
                // Add constraint that operand2 > operand1 (for SUB underflow)
                {
                    id: 'overflow_check',
                    type: 'assert',
                    expr: {
                        type: 'expression',
                        operator: 'GT',
                        operands: [record.operand2, record.operand1],
                        size: 256
                    }
                }
            ]);
            
            if (canOverflow) {
                const exploit = await this.generateExploit(path);
                
                vulnerabilities.push({
                    type: 'UNDERFLOW',
                    severity: 'HIGH',
                    pc: record.pc,
                    description: `Integer underflow at PC ${record.pc}. Subtraction can result in negative value wrapping to large positive.`,
                    path: path.constraints,
                    exploit
                });
            }
        }
        
        return vulnerabilities;
    }
    
    private async detectSelfDestruct(path: ExecutionPath): Promise<Vulnerability[]> {
        const vulnerabilities: Vulnerability[] = [];
        
        // Check if path ends with SELFDESTRUCT
        if (path.status === 'completed') {
            // Check path constraints - can anyone reach this?
            const model = await this.solver.getModel(path.constraints);
            
            if (model) {
                vulnerabilities.push({
                    type: 'SELFDESTRUCT',
                    severity: 'HIGH',
                    pc: path.state.pc,
                    description: 'SELFDESTRUCT can be triggered',
                    path: path.constraints,
                    exploit: this.modelToExploit(model)
                });
            }
        }
        
        return vulnerabilities;
    }
    
    private async generateExploit(path: ExecutionPath): Promise<ConcreteInput | undefined> {
        const model = await this.solver.getModel(path.constraints);
        
        if (!model) return undefined;
        
        return this.modelToExploit(model);
    }
    
    private modelToExploit(model: Map<string, bigint>): ConcreteInput {
        // Extract calldata values from model
        const calldataValues: bigint[] = [];
        for (const [name, value] of model) {
            const match = name.match(/calldata\[(\d+)\]/);
            if (match) {
                const idx = parseInt(match[1]);
                calldataValues[idx] = value;
            }
        }
        
        // Convert to bytes
        const calldata = new Uint8Array(calldataValues.length * 32);
        for (let i = 0; i < calldataValues.length; i++) {
            const value = calldataValues[i] || 0n;
            for (let j = 0; j < 32; j++) {
                calldata[i * 32 + (31 - j)] = Number((value >> BigInt(j * 8)) & 0xFFn);
            }
        }
        
        return {
            calldata,
            value: model.get('callvalue') || 0n,
            caller: model.get('caller') || 0n
        };
    }
}
```

# Chapter 9: The Solidity On-Chain Implementation

## 9.1 Design Considerations

**Why build a symbolic executor in Solidity?**

1. **On-chain verification:** Prove contract safety without trusting off-chain tools
2. **Composability:** Other contracts can call it for runtime checks
3. **Trustless auditing:** Verification results are cryptographically committed

**Limitations:**
- Gas costs limit path exploration
- No floating point (not an issue for EVM analysis)
- Memory constraints
- Can't use Z3 directly (need simplified constraint solving)

## 9.2 Architecture

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

/**
 * @title HeimdallOnChain
 * @notice On-chain symbolic execution engine for EVM bytecode analysis
 */
contract HeimdallOnChain {
    
    // ==================== Type Definitions ====================
    
    /**
     * @notice Represents a symbolic value
     */
    struct SymbolicValue {
        ValueType valueType;
        uint256 concreteValue;    // If concrete
        bytes32 symbolicId;       // If symbolic (unique identifier)
        bytes expressionData;     // If expression (encoded operation)
    }
    
    enum ValueType { Concrete, Symbolic, Expression }
    
    /**
     * @notice Represents the execution state
     */
    struct ExecutionState {
        uint256 pc;               // Program counter
        SymbolicValue[] stack;    // EVM stack
        mapping(uint256 => SymbolicValue) memory_;  // Memory
        mapping(bytes32 => SymbolicValue) storage_; // Storage
        bytes32[] constraints;    // Path constraints
        bool reverted;
    }
    
    /**
     * @notice Detected vulnerability
     */
    struct Vulnerability {
        VulnType vulnType;
        uint256 pc;
        string description;
    }
    
    enum VulnType { Reentrancy, Overflow, AccessControl, SelfDestruct }
    
    // ==================== State Variables ====================
    
    uint256 public constant MAX_PATHS = 100;
    uint256 public constant MAX_DEPTH = 50;
    uint256 public constant MAX_STACK_SIZE = 1024;
    
    bytes public targetBytecode;
    mapping(uint256 => bool) public jumpDests;
    
    // ==================== Main Functions ====================
    
    /**
     * @notice Analyze bytecode for vulnerabilities
     * @param bytecode The contract bytecode to analyze
     * @return vulnerabilities Array of detected vulnerabilities
     */
    function analyze(bytes calldata bytecode) 
        external 
        returns (Vulnerability[] memory vulnerabilities) 
    {
        // Store bytecode
        targetBytecode = bytecode;
        
        // Find JUMPDEST locations
        _findJumpDests(bytecode);
        
        // Create initial state
        ExecutionState memory initialState;
        initialState.pc = 0;
        initialState.stack = new SymbolicValue[](0);
        initialState.reverted = false;
        
        // Execute symbolically
        vulnerabilities = _symbolicExecute(initialState, bytecode);
    }
    
    /**
     * @notice Check a specific property
     * @param bytecode Contract bytecode
     * @param propertySelector Property to check (function selector)
     */
    function checkProperty(
        bytes calldata bytecode,
        bytes4 propertySelector
    ) external returns (bool safe, string memory reason) {
        // Set up execution with specific property check
        // ...
    }
    
    // ==================== Internal Functions ====================
    
    function _findJumpDests(bytes memory bytecode) internal {
        uint256 pc = 0;
        while (pc < bytecode.length) {
            uint8 opcode = uint8(bytecode[pc]);
            
            if (opcode == 0x5B) {  // JUMPDEST
                jumpDests[pc] = true;
            }
            
            // Skip PUSH operands
            if (opcode >= 0x60 && opcode <= 0x7F) {
                pc += uint256(opcode) - 0x60 + 2;
            } else {
                pc += 1;
            }
        }
    }
    
    function _symbolicExecute(
        ExecutionState memory state,
        bytes memory bytecode
    ) internal returns (Vulnerability[] memory) {
        Vulnerability[] memory vulns = new Vulnerability[](10);
        uint256 vulnCount = 0;
        
        // Execution queue (simplified - DFS in practice)
        ExecutionState[] memory worklist = new ExecutionState[](MAX_PATHS);
        worklist[0] = state;
        uint256 worklistSize = 1;
        
        while (worklistSize > 0) {
            // Get next state
            worklistSize--;
            ExecutionState memory current = worklist[worklistSize];
            
            // Execute one step
            (
                ExecutionState memory nextState,
                ExecutionState memory branchState,
                bool hasBranch,
                Vulnerability memory vuln,
                bool hasVuln
            ) = _executeStep(current, bytecode);
            
            // Record vulnerability
            if (hasVuln) {
                vulns[vulnCount++] = vuln;
            }
            
            // Add states to worklist
            if (!nextState.reverted && nextState.pc < bytecode.length) {
                worklist[worklistSize++] = nextState;
            }
            
            if (hasBranch && !branchState.reverted && worklistSize < MAX_PATHS) {
                worklist[worklistSize++] = branchState;
            }
        }
        
        // Resize result array
        Vulnerability[] memory result = new Vulnerability[](vulnCount);
        for (uint256 i = 0; i < vulnCount; i++) {
            result[i] = vulns[i];
        }
        
        return result;
    }
    
    function _executeStep(
        ExecutionState memory state,
        bytes memory bytecode
    ) internal view returns (
        ExecutionState memory nextState,
        ExecutionState memory branchState,
        bool hasBranch,
        Vulnerability memory vuln,
        bool hasVuln
    ) {
        nextState = state;
        hasBranch = false;
        hasVuln = false;
        
        if (state.pc >= bytecode.length) {
            return (nextState, branchState, hasBranch, vuln, hasVuln);
        }
        
        uint8 opcode = uint8(bytecode[state.pc]);
        
        // Handle opcodes
        if (opcode == 0x00) {
            // STOP
            nextState.reverted = true;  // Mark as complete
        }
        else if (opcode == 0x01) {
            // ADD
            require(state.stack.length >= 2, "Stack underflow");
            SymbolicValue memory a = state.stack[state.stack.length - 1];
            SymbolicValue memory b = state.stack[state.stack.length - 2];
            
            _stackPop(nextState);
            _stackPop(nextState);
            
            if (a.valueType == ValueType.Concrete && b.valueType == ValueType.Concrete) {
                _stackPush(nextState, SymbolicValue({
                    valueType: ValueType.Concrete,
                    concreteValue: a.concreteValue + b.concreteValue,
                    symbolicId: bytes32(0),
                    expressionData: ""
                }));
            } else {
                // Create symbolic expression
                _stackPush(nextState, SymbolicValue({
                    valueType: ValueType.Expression,
                    concreteValue: 0,
                    symbolicId: keccak256(abi.encode("ADD", a, b)),
                    expressionData: abi.encode("ADD", a, b)
                }));
            }
            
            nextState.pc++;
        }
        else if (opcode == 0x03) {
            // SUB
            require(state.stack.length >= 2, "Stack underflow");
            SymbolicValue memory a = state.stack[state.stack.length - 1];
            SymbolicValue memory b = state.stack[state.stack.length - 2];
            
            _stackPop(nextState);
            _stackPop(nextState);
            
            // Check for underflow
            if (b.valueType == ValueType.Symbolic || 
                (a.valueType == ValueType.Concrete && 
                 b.valueType == ValueType.Concrete && 
                 b.concreteValue > a.concreteValue)) {
                hasVuln = true;
                vuln = Vulnerability({
                    vulnType: VulnType.Overflow,
                    pc: state.pc,
                    description: "Potential integer underflow in SUB"
                });
            }
            
            if (a.valueType == ValueType.Concrete && b.valueType == ValueType.Concrete) {
                _stackPush(nextState, SymbolicValue({
                    valueType: ValueType.Concrete,
                    concreteValue: a.concreteValue - b.concreteValue,
                    symbolicId: bytes32(0),
                    expressionData: ""
                }));
            } else {
                _stackPush(nextState, SymbolicValue({
                    valueType: ValueType.Expression,
                    concreteValue: 0,
                    symbolicId: keccak256(abi.encode("SUB", a, b)),
                    expressionData: abi.encode("SUB", a, b)
                }));
            }
            
            nextState.pc++;
        }
        else if (opcode == 0x57) {
            // JUMPI - conditional jump, FORK execution
            require(state.stack.length >= 2, "Stack underflow");
            SymbolicValue memory dest = state.stack[state.stack.length - 1];
            SymbolicValue memory condition = state.stack[state.stack.length - 2];
            
            _stackPop(nextState);
            _stackPop(nextState);
            
            // True branch (take jump)
            if (dest.valueType == ValueType.Concrete) {
                uint256 target = dest.concreteValue;
                if (jumpDests[target]) {
                    hasBranch = true;
                    branchState = _cloneState(state);
                    _stackPop(branchState);
                    _stackPop(branchState);
                    branchState.pc = target;
                    // Add constraint: condition != 0
                }
            }
            
            // False branch (fall through)
            nextState.pc++;
            // Add constraint: condition == 0
        }
        else if (opcode == 0xF1) {
            // CALL - check for reentrancy
            hasVuln = true;
            vuln = Vulnerability({
                vulnType: VulnType.Reentrancy,
                pc: state.pc,
                description: "External CALL detected - potential reentrancy"
            });
            
            // Pop 7 arguments, push 1 result
            for (uint i = 0; i < 7; i++) {
                _stackPop(nextState);
            }
            _stackPush(nextState, SymbolicValue({
                valueType: ValueType.Symbolic,
                concreteValue: 0,
                symbolicId: keccak256(abi.encode("CALL_RESULT", state.pc)),
                expressionData: ""
            }));
            
            nextState.pc++;
        }
        else if (opcode >= 0x60 && opcode <= 0x7F) {
            // PUSH1 - PUSH32
            uint256 n = uint256(opcode) - 0x60 + 1;
            uint256 value = 0;
            
            for (uint256 i = 0; i < n; i++) {
                if (state.pc + 1 + i < bytecode.length) {
                    value = (value << 8) | uint8(bytecode[state.pc + 1 + i]);
                }
            }
            
            _stackPush(nextState, SymbolicValue({
                valueType: ValueType.Concrete,
                concreteValue: value,
                symbolicId: bytes32(0),
                expressionData: ""
            }));
            
            nextState.pc += n + 1;
        }
        else {
            // Other opcodes - skip for brevity
            nextState.pc++;
        }
        
        return (nextState, branchState, hasBranch, vuln, hasVuln);
    }
    
    // Stack helpers
    function _stackPush(ExecutionState memory state, SymbolicValue memory value) internal pure {
        SymbolicValue[] memory newStack = new SymbolicValue[](state.stack.length + 1);
        for (uint i = 0; i < state.stack.length; i++) {
            newStack[i] = state.stack[i];
        }
        newStack[state.stack.length] = value;
        state.stack = newStack;
    }
    
    function _stackPop(ExecutionState memory state) internal pure {
        require(state.stack.length > 0, "Stack underflow");
        SymbolicValue[] memory newStack = new SymbolicValue[](state.stack.length - 1);
        for (uint i = 0; i < newStack.length; i++) {
            newStack[i] = state.stack[i];
        }
        state.stack = newStack;
    }
    
    function _cloneState(ExecutionState memory state) internal pure returns (ExecutionState memory) {
        ExecutionState memory cloned;
        cloned.pc = state.pc;
        cloned.stack = new SymbolicValue[](state.stack.length);
        for (uint i = 0; i < state.stack.length; i++) {
            cloned.stack[i] = state.stack[i];
        }
        cloned.reverted = state.reverted;
        return cloned;
    }
}
```
